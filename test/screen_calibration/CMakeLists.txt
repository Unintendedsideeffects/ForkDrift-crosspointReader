cmake_minimum_required(VERSION 3.16)
project(ScreenCalibrationTest LANGUAGES CXX)
enable_testing()
set(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_executable(screen_calibration ScreenCalibrationTest.cpp ${REPO_ROOT}/lib/ScreenCalibration/HalScreenCalibration.cpp)
target_include_directories(screen_calibration PRIVATE stubs ${REPO_ROOT}/lib/ScreenCalibration)
target_compile_features(screen_calibration PRIVATE cxx_std_17)
target_compile_options(screen_calibration PRIVATE -Wall -Wextra -pedantic -UNDEBUG)
add_test(NAME screen_calibration COMMAND screen_calibration)
